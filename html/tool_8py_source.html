113  index=i
114  break

120  innerVersion='0.00000'+innerVersion
121  else:
122  innerVersion='0.0000'+innerVersion #为了确保a10<b1
123  elif Type=='b':#beta版
124  if len(innerVersion)==1:
125  innerVersion='0.000'+innerVersion
126  else:
127  innerVersion='0.00'+innerVersion #为了确保b10<rc1
128  elif Type=='rc': #release版
129  if len(innerVersion)==1:
130  innerVersion='0.0'+innerVersion
131  else:
132  innerVersion='0.'+innerVersion

135  lst=version[0:index].split('.')
136  if len(lst)==2:
137  lst.append('0')

139  lst[1]='0'+lst[1]
