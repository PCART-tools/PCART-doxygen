58 def applyEdits(sourceBytes,edits):
59  errLst=[]
60  try:
61  source=sourceBytes.decode('utf-8-sig')
62  root=ast.parse(source)
63  lines=source.splitlines(keepends=True)
64  byteLines=sourceBytes.splitlines(keepends=True)
65  lineOffsets=[]
66  offset=0
67  for line in byteLines:
68  lineOffsets.append(offset)
69  offset+=len(line)
70  if sourceBytes.startswith(b'\xef\xbb\xbf'):
71  lineOffsets[0]+=3
72  callNodes={(node.lineno,node.col_offset,node.end_lineno,node.end_col_offset):node
73  for node in ast.walk(root) if isinstance(node,ast.Call)}
74  commentOffsets=[]
75  for token in tokenize.generate_tokens(io.StringIO(source).readline):
76  if token.type==tokenize.COMMENT:
77  line,column=token.start
78  commentOffsets.append(lineOffsets[line-1]+len(lines[line-1][:column].encode('utf-8')))
79  except (UnicodeError,SyntaxError,ValueError,tokenize.TokenError) as e:
80  return sourceBytes,[f'Patch skipped: cannot parse source: {e}\n']
81 
82  candidates=[]
83  for edit in edits:
84  callId=edit.get('id','unknown')
85  try:
86  span=tuple(edit.get(key) for key in ('lineno','col_offset','end_lineno','end_col_offset'))
87  if span not in callNodes:
88  raise ValueError('missing or invalid callsite span')
89  original=ast.parse(edit['call_text'],mode='eval').body
90  if ast.dump(original)!=ast.dump(callNodes[span]):
91  raise ValueError('original call does not match the source span')
92  fixed=ast.parse(edit['fixed_api'],mode='eval').body
93  if not isinstance(fixed,ast.Call):
94  raise ValueError('repair result is not a single call expression')
95  start=lineOffsets[span[0]-1]+span[1]
96  end=lineOffsets[span[2]-1]+span[3]
97  if any(start<=offset<end for offset in commentOffsets):
98  raise ValueError('callsite contains comments that replacement would remove')
99  candidates.append((start,end,edit))
100  except (KeyError,TypeError,SyntaxError,ValueError) as e:
101  errLst.append(f'Patch skipped for {callId}: {e}\n')
102 
103  candidates.sort(key=lambda item:item[0])
104  overlaps=set()
105  for index,(start,end,edit) in enumerate(candidates):
106  for other in range(index+1,len(candidates)):
107  if candidates[other][0]>=end:
108  break
109  overlaps.update((index,other))
110  modified=sourceBytes
111  newline='\r\n' if b'\r\n' in sourceBytes else '\n'
112  for index in range(len(candidates)-1,-1,-1):
113  start,end,edit=candidates[index]
114  if index in overlaps:
115  errLst.append(f"Patch skipped for {edit['id']}: overlapping callsite spans\n")
116  continue
117  replacement=edit['fixed_api'].replace('\r\n','\n').replace('\n',newline).encode('utf-8')
118  modified=modified[:start]+replacement+modified[end:]
119  try:
120  ast.parse(modified.decode('utf-8-sig'))
121  except (SyntaxError,ValueError) as e:
122  errLst.append(f'Patch skipped: modified file has invalid syntax: {e}\n')
123  return sourceBytes,errLst
124  return modified,errLst
